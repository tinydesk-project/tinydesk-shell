cmake_minimum_required(VERSION 3.16)
option(TDSH_BUILD_HOST "Build the POSIX host port and tests instead of the ESP32-C6 firmware" OFF)
if(TDSH_BUILD_HOST)
    include("${CMAKE_CURRENT_LIST_DIR}/cmake/host.cmake")
    return()
endif()

if(NOT DEFINED ENV{IDF_PATH})
    message(FATAL_ERROR "Select ESP-IDF 5.3.1 in VS Code and open an ESP-IDF terminal.")
endif()
set(IDF_TARGET "esp32c6" CACHE STRING "TinyDesk Shell firmware target (projects/esp32 for the classic ESP32, projects/esp32s3 for the ESP32-S3)")
if(NOT IDF_TARGET STREQUAL "esp32c6")
    message(FATAL_ERROR "This project is for the ESP32-C6 (idf.py set-target esp32c6); the classic ESP32 one is projects/esp32, the ESP32-S3 one projects/esp32s3.")
endif()
file(STRINGS "${CMAKE_CURRENT_LIST_DIR}/VERSION" PROJECT_VER LIMIT_COUNT 1)
# Configure again when VERSION changes (the version is read only then).
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/VERSION")
set(TDSH_SDK_ROOT "${CMAKE_CURRENT_LIST_DIR}")
list(APPEND EXTRA_COMPONENT_DIRS "${TDSH_SDK_ROOT}/ports/esp_idf/components")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(tdsh_c6)
include("${TDSH_SDK_ROOT}/ports/esp_idf/tdsh_espidf_project_setup.cmake")
